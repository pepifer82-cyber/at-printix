+var
+  List: TStringList;
+  i, sep, rcode: Integer;
+  item, un, path: String;
+  launched: Boolean;
+begin
+  Result := '';
+  if not FullCleanRequested then Exit;
+
+  List := TStringList.Create;
+  try
+    List.Sorted := True;
+    List.Duplicates := dupIgnore;
+    ScanPreviousAandT(HKEY_LOCAL_MACHINE, List);
+    ScanPreviousAandT(HKEY_CURRENT_USER, List);
+
+    // Retira cada desinstalador registrado de A&T, incluido el AppId de la linea 0.5.x.
+    for i := 0 to List.Count - 1 do
+    begin
+      item := List[i];
+      sep := Pos('|', item);
+      if sep <= 0 then Continue;
+      un := Copy(item, sep + 1, Length(item));
+      path := StripOuterQuotes(un);
+      if path = '' then Continue;
+
+      launched := Exec(path, '/VERYSILENT /FULLCLEAN /NORESTART /SUPPRESSMSGBOXES',
+                       '', SW_HIDE, ewWaitUntilTerminated, rcode);
+      if (not launched) or (rcode <> 0) then
+      begin
+        Result := 'No se pudo desinstalar la versión anterior de A&T (' + Copy(item, 1, sep - 1) + '). Código: ' + IntToStr(rcode) + '. La instalación nueva se canceló para evitar mezclar archivos.';
+        Exit;
+      end;
+    end;
+  finally
+    List.Free;
+  end;
+
+  // Cierra cualquier proceso residual y borra TODOS los datos locales A&T
+  // en los perfiles Windows. No se toca el proyecto ni las licencias de Supabase.
+  Exec(ExpandConstant('{sys}\taskkill.exe'), '/IM ATPrinterAgent.exe /T /F',
+       '', SW_HIDE, ewWaitUntilTerminated, rcode);
